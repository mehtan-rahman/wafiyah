\bab{الباب السابع العشر في \com{يمد}{كذا في الأصل، ولعل الصواب: مدّ يمدّ} (مضاعف)}
% Layout: six columns of continuous paradigms (see p.31 note). Col. 1 is the
% صرف صغير of مدّ: the uncontracted forms (مَدَدَ …) above a rule, the
% contracted forms (مَدَّ …) below it. Each boxed heading holds the title and the
% uncontracted base form, given here as two consecutive rows.
\begin{sarf}{|c|c|c|c|c|c|}
\thead{نصر} & \thead{ماضي معروف} & تَمُدُّ & مُدَّ & يُمْدَدْنَ & مُدُّوْا\\
مَدَدَ & مَدَدَ & تَمُدَّانِ & مُدَّا & تُمَدُّ & مُدِّيْ\\
يَمْدُدُ & مَدَّ & يَمْدُدْنَ & مُدُّوْا & تُمَدَّانِ & مُدَّا\\
مَدْدًا & مَدَّا & تَمُدُّ & مُدَّتْ & تُمَدُّوْنَ & اُمْدُدْنَ\\
مَادِدٌ & مَدُّوْا & تَمُدَّانِ & مُدَّتَا & تُمَدِّيْنَ & \thead{نهي حاضر}\\
مُدِدَ & مَدَّتْ & تَمُدُّوْنَ & مُدِدْنَ & تُمَدَّانِ & لَا تَمْدُدْ\\
يُمْدَدُ & مَدَّتَا & تَمُدِّيْنَ & مُدِدْتَ & تُمْدَدْنَ & لَا تَمُدَّ\\
مَدْدًا & مَدَدْنَ & تَمُدَّانِ & مُدِدْتُمَا & اُمَدُّ & لَا تَمُدَّا\\
مَمْدُوْدٌ & مَدَدْتَ & تَمْدُدْنَ & مُدِدْتُمْ & نُمَدُّ & لَا تَمُدُّوْا\\
اُمْدُدْ & مَدَدْتُمَا & اَمُدُّ & مُدِدْتِ & \thead{اسم مفعول} & لَا تَمُدِّيْ\\
لَا تَمْدُدْ & مَدَدْتُمْ & نَمُدُّ & مُدِدْتُمَا & مَمْدُوْدٌ & لَا تَمُدَّا\\
مَدَّ & مَدَدْتِ & \thead{اسم فاعل} & مُدِدْتُنَّ & مَمْدُوْدٌ & لَا تَمْدُدْنَ\\
يَمُدُّ & مَدَدْتُمَا & مَادِدٌ & مُدِدْتُ & مَمْدُوْدَانِ & \thead{مضارع بلم}\\
مَدًّا & مَدَدْتُنَّ & مَادٌّ & مُدِدْنَا & مَمْدُوْدُوْنَ & لَمْ يَمْدُدْ\\
مَادٌّ & مَدَدْتُ & مَادَّانِ & \thead{مضارع مجهول} & مَمْدُوْدَةٌ & لَمْ يَمُدَّ\\
مُدَّ & مَدَدْنَا & مَادُّوْنَ & يُمْدَدُ & مَمْدُوْدَتَانِ & لَمْ يَمُدَّا\\
يُمَدُّ & \thead{مضارع معروف} & مَادَّةٌ & يُمَدُّ & مَمْدُوْدَاتٌ & لَمْ يَمُدُّوْا\\
مَدًّا & يَمْدُدُ & مَادَّتَانِ & يُمَدَّانِ & \thead{أمر حاضر} & لَمْ تَمُدَّ\\
مَمْدُوْدٌ & يَمُدُّ & مَادَّاتٌ & يُمَدُّوْنَ & اُمْدُدْ & لَمْ تَمُدَّا\\
مُدَّ & يَمُدَّانِ & \thead{ماضي مجهول} & تُمَدُّ & مُدَّ & لَمْ يَمْدُدْنَ\\
لَا تَمُدَّ & يَمُدُّوْنَ & مُدِدَ & تُمَدَّانِ & مُدَّا & لَمْ تَمُدَّ\\
\end{sarf}
