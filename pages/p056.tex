\bab{الباب الثاني والعشرون في الأفعال المركبة}
\begin{sarf}{|c|c|c|c|}
\thead{ماضي بعيد} & \thead{ماضي بعيد استمراري} & \thead{ماضي بعيد استمراري} & \thead{ماضي ابتدائي}\\
{\urdufont وہ کیا تھا} & {\urdufont وہ کرتا تھا} & {\urdufont وہ نہیں کرتا تھا} & {\urdufont وہ کرنے لگا}\\\hline
كَانَ فَعَلَ & كَانَ يَفْعَلُ & لَمْ يَكُنْ يَفْعَلُ & جَعَلَ يَفْعَلُ\\
كَانَا فَعَلَا & كَانَا يَفْعَلَانِ & لَمْ يَكُوْنَا يَفْعَلَانِ & جَعَلَا يَفْعَلَانِ\\
كَانُوْا فَعَلُوْا & كَانُوْا يَفْعَلُوْنَ & لَمْ يَكُوْنُوْا يَفْعَلُوْنَ & جَعَلُوْا يَفْعَلُوْنَ\\
كَانَتْ فَعَلَتْ & كَانَتْ تَفْعَلُ & لَمْ تَكُنْ تَفْعَلُ & جَعَلَتْ تَفْعَلُ\\
كَانَتَا فَعَلَتَا & كَانَتَا تَفْعَلَانِ & لَمْ تَكُوْنَا تَفْعَلَانِ & جَعَلَتَا تَفْعَلَانِ\\
كُنَّ فَعَلْنَ & كُنَّ يَفْعَلْنَ & لَمْ \unclear{يَكُنَّ} يَفْعَلْنَ & جَعَلْنَ يَفْعَلْنَ\\
كُنْتَ فَعَلْتَ & كُنْتَ تَفْعَلُ & لَمْ تَكُنْ تَفْعَلُ & جَعَلْتَ تَفْعَلُ\\
كُنْتُمَا فَعَلْتُمَا & كُنْتُمَا تَفْعَلَانِ & لَمْ تَكُوْنَا تَفْعَلَانِ & جَعَلْتُمَا تَفْعَلَانِ\\
كُنْتُمْ فَعَلْتُمْ & كُنْتُمْ تَفْعَلُوْنَ & لَمْ تَكُوْنُوْا تَفْعَلُوْنَ & جَعَلْتُمْ تَفْعَلُوْنَ\\
\end{sarf}
\begin{sarf}{|c|c|c|c|}
\thead{ماضي المقاربة} & \thead{ماضي المحال} & \thead{ماضي استمراري} & \thead{مستقبل استمراري}\\
{\urdufont وہ کرنے کے قریب ہوا} & {\urdufont وہ کرنے کے قریب بھی نہیں ہوا} & {\urdufont وہ کرتا رہا} & {\urdufont جب تک کرتا رہیگا}\\\hline
كَادَ يَفْعَلُ & لَمْ يَكَدْ يَفْعَلُ & لَمْ يَزَلْ يَفْعَلُ & مَا دَامَ يَفْعَلُ\\
كَادَا يَفْعَلَانِ & لَمْ يَكَادَا يَفْعَلَانِ & لَمْ يَزَالَا يَفْعَلَانِ & مَا دَامَا يَفْعَلَانِ\\
كَادُوْا يَفْعَلُوْنَ & لَمْ يَكَادُوْا يَفْعَلُوْنَ & لَمْ يَزَالُوْا يَفْعَلُوْنَ & مَا دَامُوْا يَفْعَلُوْنَ\\
كَادَتْ تَفْعَلُ & لَمْ تَكَدْ تَفْعَلُ & لَمْ تَزَلْ تَفْعَلُ & مَا دَامَتْ تَفْعَلُ\\
كَادَتَا تَفْعَلَانِ & لَمْ تَكَادَا تَفْعَلَانِ & لَمْ تَزَالَا تَفْعَلَانِ & مَا دَامَتَا تَفْعَلَانِ\\
كِدْنَ يَفْعَلْنَ & لَمْ يَكَدْنَ يَفْعَلْنَ & لَمْ يَزَلْنَ يَفْعَلْنَ & مَا دُمْنَ يَفْعَلْنَ\\
كِدْتَ تَفْعَلُ & لَمْ تَكَدْ تَفْعَلُ & لَمْ تَزَلْ تَفْعَلُ & مَا دُمْتَ تَفْعَلُ\\
كِدْتُمَا تَفْعَلَانِ & لَمْ تَكَادَا تَفْعَلَانِ & لَمْ تَزَالَا تَفْعَلَانِ & مَا دُمْتُمَا تَفْعَلَانِ\\
\end{sarf}
